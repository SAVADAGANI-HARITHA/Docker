---
apiVersion: v1
kind: Pod
metadata:
  name: pod-1
spec:
  containers:
    - name: cont2
      image: nginx
      ports:
        - containerPort: 80
      args:
      - /bin/sh
      - -c
      - touch /tmp/mustafa; sleep 1000;
      livenessProbe:
        exec:
          command:
          - cat
          - /tmp/mustafa
        initialDelaySeconds: 5
        periodSeconds: 10
        timeoutSeconds: 30
