  name: flm
spec:
  replicas: 2
  selector:
    matchLabels:
      app: swiggy
  template:
    metadata:
      labels:
        app: swiggy

    spec:
      containers:
        - name: cont1
          image: shaikmustafa/paytm:bus
          ports:
            - containerPort: 80
       tolerations:
        - key: dedicated
          operator: Equal
          value: database
          effect: NoSchedule

apiVersion: storage.k8s.io/v1
kind: StorageClass
metadata:
  name: mysql-gp3
  annotations:
    storageclass.kubernetes.io/is-default-class: "false"
provisioner: ebs.csi.aws.com
parameters:
  type: gp3
  fsType: ext4
  encrypted: "true"
  iops: "3000"
  throughput: "125"
reclaimPolicy: Retain
allowVolumeExpansion: true
volumeBindingMode: WaitForFirstConsumer

            claimName: mysql-data
